## Hash

手写unordered\_map

```
const int SZ = 1428571;
struct hash_map {
  struct data {
    int u;
    int v, nex;
  };               
  data e[SZ << 1]; 
  int h[SZ], cnt;
  inline int hash(int u) { return u % SZ; }
  int& operator[](int u) {
    int hu = hash(u); 
    for (int i = h[hu]; i; i = e[i].nex)
      if (e[i].u == u) return e[i].v;
    return e[++cnt] = (data){u, 0, h[hu]}, h[hu] = cnt, e[cnt].v;
  }
  int get(int u) {
    int hu = hash(u); 
    for (int i = h[hu]; i; i = e[i].nex){
        if (e[i].u == u) return e[i].v;
    }
    return 0;
  }
  hash_map() {
    cnt = 0;
    memset(h, 0, sizeof(h));
  }
}cnt;
```
