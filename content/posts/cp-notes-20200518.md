一道水题，一开始还想写一个从最高位逐位比对的cmp……

真是被秀到

```
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e5 + 7;
string a[N];
bool cmp(string a, string b) { return a + b > b + a; }
int main() {
    int n;cin >> n;
    for (int i = 0; i < n; ++i) cin >> a[i];
    sort(a, a + n, cmp);
    for (int i = 0; i < n; ++i) cout << a[i];
    return 0;
}
```
